% 3. Context și Obiective
\begin{frame}{Contextul și Obiectivele Proiectului}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \textbf{Limitările sistemelor actuale:}
            \begin{itemize}
                \item Absența titlurilor Tower Defense cu asimetrie funcțională completă (Atacator vs. Apărător).
                \item Vulnerabilități structurale de securitate (cheating) în topologiile de rețea Peer-to-Peer.
                \item Desincronizarea stării partajate sub influența latenței.
            \end{itemize}

            \textit{\small Titluri de referință din domeniu:}\\

% 10. Testare și Evaluare
\begin{frame}{Testare și Evaluare}
    \begin{block}{Validarea Concurenței Backend (xUnit \& NSubstitute)}
        \begin{itemize}
            \item \textbf{Thread-Safety:} Rularea a 1000 de cereri asincrone simultane (\texttt{Task.WhenAll}) a generat exact 500 de meciuri valide, confirmând prevenirea fenomenului de \textit{race conditions} prin \texttt{lock}.
            \item \textbf{Idempotență și Izolare:} Gestionarea deterministă a structurilor \texttt{HashSet} prin respingerea apelurilor redundante și curățarea referințelor la abandon.
        \end{itemize}
    \end{block}

    \begin{block}{Profilare Funcțională Client-Side (DeckService)}
        \begin{itemize}
            \item \textbf{Eficiența Debouncing-ului:} Un test cu 20 de acțiuni rapide în interfață a comasat execuția în doar 2 apeluri HTTP înregistrate de backend, reducând traficul de rețea cu 90\%.
            \item \textbf{Reziliența la Latență:} Inserarea unor întârzieri artificiale a confirmat capacitatea algoritmului de a respinge actualizările vizuale perimate, validând starea prin \texttt{\_latestVersion}.
        \end{itemize}
    \end{block}
\end{frame}
