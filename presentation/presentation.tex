% 2. Context și Obiective
\begin{frame}{Contextul și Obiectivele Proiectului}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \textbf{Problema identificată:}
            \begin{itemize}
                \item Lipsa jocurilor Tower Defense cu asimetrie reală (Defender vs Attacker).
                \item Vulnerabilitatea la trișat în jocurile Peer-to-Peer.
                \item Latența critică în sincronizarea stării jocului.
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Soluția propusă:}
            \begin{itemize}
                \item Implementarea unui model \textbf{Server-Authoritative} complet.
                \item Sistem de matchmaking robust bazat pe ASP.NET Core.
                \item Arhitectură modulară în Godot Engine (C\#).
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

% 8. Optimizări: Sincronizarea Pachetului
\begin{frame}{Sincronizarea Asincronă (Deck Debouncing)}
    \begin{columns}[c]
        \begin{column}{0.65\textwidth}
            \centering
            \includegraphics[width=\textwidth, height=0.75\textheight, keepaspectratio]{images/async-debouncing-sync.png}
            \\ \vspace{1mm} \textit{\tiny Mecanismul de Debouncing}
        \end{column}
        \begin{column}{0.35\textwidth}
            \begin{itemize}
                \item \small \textbf{Problema:} Salvarea frecventă a pachetului în timpul editării.
                \item \small \textbf{Soluția:} Timer de debouncing (500ms) + comparare \texttt{VersionID}.
                \item \small \textbf{Rezultat:} Reducerea încărcării serverului cu 85-90\%.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}
